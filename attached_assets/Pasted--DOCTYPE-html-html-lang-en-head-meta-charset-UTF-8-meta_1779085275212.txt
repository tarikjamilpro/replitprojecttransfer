    <script>
        let currentPrompts = [];

        function updateAspectRatio() {
            const platform = document.getElementById('platform').value;
            const aspectSelect = document.getElementById('aspectRatio');

            const ratios = {
                "YouTube": "16:9",
                "Instagram": "9:16",
                "TikTok": "9:16",
                "Facebook Reels": "9:16",
                "Pinterest": "2:3",
                "Threads": "9:16",
                "General": "16:9"
            };

            if (ratios[platform]) {
                aspectSelect.value = ratios[platform];
            }
        }

        function generatePrompts() {
            const title = document.getElementById('titleInput').value.trim();
            if (!title) return alert("Please enter a title");

            const platform = document.getElementById('platform').value;
            const style = document.getElementById('style').value;
            const aspectRatio = document.getElementById('aspectRatio').value;

            currentPrompts = createPrompts(title, platform, style, aspectRatio);
            renderPrompts();
            document.getElementById('outputSection').classList.remove('hidden');

            // Save to history
            saveToHistory(title, currentPrompts);
        }

        function regeneratePrompts() {
            if (currentPrompts.length === 0) return;
            const title = document.getElementById('titleInput').value.trim();
            const platform = document.getElementById('platform').value;
            const style = document.getElementById('style').value;
            const aspectRatio = document.getElementById('aspectRatio').value;

            currentPrompts = createPrompts(title, platform, style, aspectRatio);
            renderPrompts();
        }

        function createPrompts(title, platform, style, aspectRatio) {
            const base = `Create a highly click-worthy thumbnail for "${title}". ${style} style, optimized for ${platform}, aspect ratio ${aspectRatio}.`;

            return [
                `${base} Dramatic lighting, strong emotion, bold composition, high contrast.`,
                `${base} Vibrant colors, powerful focal point, modern and energetic style.`,
                `${base} Minimalist, clean design, premium aesthetic, elegant lighting.`,
                `${base} Emotional storytelling, rich colors, high visual impact.`,
                `${base} Luxury feel, cinematic lighting, highly detailed and professional.`,
                `${base} Bold and scroll-stopping design, optimized for maximum engagement.`
            ];
        }

        function renderPrompts() {
            const container = document.getElementById('promptsContainer');
            container.innerHTML = '';

            currentPrompts.forEach((prompt, index) => {
                const div = document.createElement('div');
                div.className = `bg-white border rounded-3xl p-5 flex flex-col lg:flex-row gap-4`;
                div.innerHTML = `
                    <div class="flex-1">
                        <div class="text-xs text-purple-600 font-medium mb-1">PROMPT ${index + 1}</div>
                        <p class="text-gray-700">${prompt}</p>
                    </div>
                    <div>
                        <button onclick="copyPrompt(this)" class="px-5 py-2.5 bg-purple-600 text-white rounded-2xl text-sm flex items-center gap-2">
                            <i class="fa-solid fa-copy"></i> Copy
                        </button>
                    </div>
                `;
                container.appendChild(div);
            });
        }

        function copyPrompt(element) {
            const text = element.closest('.flex-col, .flex').querySelector('p').innerText;
            navigator.clipboard.writeText(text);
            element.innerHTML = `<i class="fa-solid fa-check"></i> Copied`;
            setTimeout(() => element.innerHTML = `<i class="fa-solid fa-copy"></i> Copy`, 1500);
        }

        function copyAllPrompts() {
            const text = currentPrompts.join('\n\n');
            navigator.clipboard.writeText(text);
            alert("All prompts copied!");
        }

        function exportAsTxt() {
            if (currentPrompts.length === 0) return;
            const text = currentPrompts.join('\n\n---\n\n');
            const blob = new Blob([text], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'thumbnail-prompts.txt';
            a.click();
        }

        // History using localStorage
        function saveToHistory(title, prompts) {
            let history = JSON.parse(localStorage.getItem('thumbnailHistory') || '[]');
            history.unshift({ title, prompts, date: new Date().toLocaleString() });
            if (history.length > 5) history.pop();
            localStorage.setItem('thumbnailHistory', JSON.stringify(history));
        }

        function showHistory() {
            const history = JSON.parse(localStorage.getItem('thumbnailHistory') || '[]');
            if (history.length === 0) return alert("No history yet");

            let html = `<div class="fixed inset-0 bg-black/50 flex items-center justify-center z-50" onclick="this.remove()">
                <div class="bg-white w-full max-w-2xl mx-4 rounded-3xl p-6" onclick="event.stopImmediatePropagation()">
                    <h3 class="text-xl font-semibold mb-4">Prompt History</h3>
                    <div class="space-y-4 max-h-[400px] overflow-auto">`;

            history.forEach((item, i) => {
                html += `<div class="border p-4 rounded-2xl">
                    <div class="font-medium">${item.title}</div>
                    <div class="text-xs text-gray-500">${item.date}</div>
                </div>`;
            });

            html += `</div></div></div>`;
            document.body.insertAdjacentHTML('beforeend', html);
        }

        // Auto update aspect ratio
        document.getElementById('platform').addEventListener('change', updateAspectRatio);
    </script>
