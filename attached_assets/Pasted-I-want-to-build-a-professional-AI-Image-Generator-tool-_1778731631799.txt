    <script>
        let currentAspectRatio = { ratio: '1:1', width: 1024, height: 1024 };

        function initializeTailwind() {
            console.log('%c[AI Image Generator] Initialized', 'color:#64748b');
        }

        function updateSliderValue(type) {
            const valueEl = document.getElementById(type + 'Value');
            const input = document.getElementById(type);
            
            if (type === 'guidance') {
                valueEl.textContent = parseFloat(input.value).toFixed(1);
            } else {
                valueEl.textContent = input.value;
            }
        }

        function selectAspectRatio(element, ratio, width, height) {
            document.querySelectorAll('#aspectRatioGroup button').forEach(btn => {
                btn.classList.remove('active', 'bg-indigo-600', 'border-indigo-500');
                btn.classList.add('border-zinc-700');
            });
            
            element.classList.add('active', 'bg-indigo-600', 'border-indigo-500');
            element.classList.remove('border-zinc-700');
            
            currentAspectRatio = { ratio, width, height };
        }

        function toggleAdvanced() {
            const settings = document.getElementById('advancedSettings');
            const icon = document.getElementById('advancedIcon');
            
            if (settings.classList.contains('hidden')) {
                settings.classList.remove('hidden');
                icon.style.transform = 'rotate(180deg)';
            } else {
                settings.classList.add('hidden');
                icon.style.transform = 'rotate(0deg)';
            }
        }

        function showToast(message, type = 'success') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            
            const bgColor = type === 'success' ? 'bg-emerald-600' : 'bg-red-600';
            const icon = type === 'success' ? 'fa-check-circle' : 'fa-exclamation-circle';
            
            toast.className = `toast flex items-center gap-x-3 px-5 py-3 rounded-2xl text-white shadow-xl ${bgColor}`;
            toast.innerHTML = `
                <i class="fas ${icon}"></i>
                <span class="text-sm font-medium">${message}</span>
            `;
            
            container.appendChild(toast);
            
            setTimeout(() => {
                toast.style.transition = 'all 0.3s ease';
                toast.style.opacity = '0';
                setTimeout(() => toast.remove(), 300);
            }, 2800);
        }

        async function generateImages() {
            const prompt = document.getElementById('prompt').value.trim();
            const negativePrompt = document.getElementById('negativePrompt').value.trim();
            const model = document.getElementById('model').value;
            const steps = parseInt(document.getElementById('steps').value);
            const guidance = parseFloat(document.getElementById('guidance').value);
            const numImages = parseInt(document.getElementById('numImages').value);

            if (!prompt) {
                showToast("Please enter a prompt", "error");
                return;
            }

            const generateBtn = document.getElementById('generateBtn');
            const btnText = document.getElementById('btnText');
            const btnIcon = document.getElementById('btnIcon');
            const imagesGrid = document.getElementById('imagesGrid');
            const emptyState = document.getElementById('emptyState');

            generateBtn.disabled = true;
            btnText.textContent = `Generating ${numImages} image${numImages > 1 ? 's' : ''}...`;
            btnIcon.className = 'fas fa-spinner fa-spin';

            if (emptyState) emptyState.style.display = 'none';
            imagesGrid.innerHTML = '';

            try {
                const response = await fetch('/api/generate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        prompt: prompt,
                        negative_prompt: negativePrompt,
                        model: model,
                        steps: steps,
                        guidance_scale: guidance,
                        num_images: numImages,
                        width: currentAspectRatio.width,
                        height: currentAspectRatio.height
                    })
                });

                if (!response.ok) {
                    const errorData = await response.json();
                    throw new Error(errorData.error || 'Failed to generate images');
                }

                const data = await response.json();

                if (!data.images || data.images.length === 0) {
                    throw new Error('No images returned from server');
                }

                data.images.forEach((imageBase64, index) => {
                    const imageCard = createImageCard(imageBase64, prompt, index);
                    imagesGrid.appendChild(imageCard);
                });

                document.getElementById('resultCount').innerHTML = `${data.images.length} image${data.images.length > 1 ? 's' : ''} generated`;
                document.getElementById('resultCount').classList.remove('hidden');

                showToast(`${data.images.length} image(s) generated successfully!`);

            } catch (error) {
                console.error(error);
                showToast(error.message || "Failed to generate images", "error");
                
                imagesGrid.innerHTML = `
                    <div class="col-span-full flex flex-col items-center justify-center py-12 text-center">
                        <i class="fas fa-exclamation-triangle text-red-400 text-4xl mb-4"></i>
                        <p class="text-red-400 font-medium">Generation Failed</p>
                        <p class="text-sm text-zinc-400 mt-1">${error.message}</p>
                    </div>
                `;
            } finally {
                generateBtn.disabled = false;
                btnText.textContent = 'Generate Images';
                btnIcon.className = 'fas fa-magic';
            }
        }

        function createImageCard(base64Image, prompt, index) {
            const card = document.createElement('div');
            card.className = `image-card bg-zinc-950 border border-zinc-800 rounded-2xl overflow-hidden group`;
            
            card.innerHTML = `
                <div class="relative aspect-square bg-black">
                    <img src="data:image/png;base64,${base64Image}" class="generated-image w-full h-full object-cover" alt="Generated image ${index + 1}">
                    
                    <div class="absolute inset-0 bg-gradient-to-t from-black/70 via-black/20 to-transparent opacity-0 group-hover:opacity-100 transition-all flex items-end p-4">
                        <div class="flex gap-2 w-full">
                            <button onclick="downloadImage('${base64Image}', ${index})" 
                                    class="flex-1 flex items-center justify-center gap-x-2 bg-white/90 hover:bg-white text-black text-xs font-semibold py-2.5 rounded-xl transition">
                                <i class="fas fa-download"></i>
                                <span>Download</span>
                            </button>
                            
                            <button onclick="copyPrompt('${prompt.replace(/'/g, "\\'")}')" 
                                    class="flex-1 flex items-center justify-center gap-x-2 bg-white/90 hover:bg-white text-black text-xs font-semibold py-2.5 rounded-xl transition">
                                <i class="fas fa-copy"></i>
                                <span>Copy Prompt</span>
                            </button>
                        </div>
                    </div>
                </div>
                
                <div class="p-3 flex items-center justify-between bg-zinc-900">
                    <div class="text-xs text-zinc-400">Image ${index + 1}</div>
                    <button onclick="regenerateImage('${prompt.replace(/'/g, "\\'")}')" 
                            class="text-xs flex items-center gap-x-1.5 text-indigo-400 hover:text-indigo-300 transition">
                        <i class="fas fa-redo text-xs"></i>
                        <span>Regenerate</span>
                    </button>
                </div>
            `;
            return card;
        }

        function downloadImage(base64, index) {
            const link = document.createElement('a');
            link.href = `data:image/png;base64,${base64}`;
            link.download = `ai-image-${Date.now()}-${index + 1}.png`;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast('Image downloaded');
        }

        function copyPrompt(promptText) {
            navigator.clipboard.writeText(promptText).then(() => {
                showToast('Prompt copied to clipboard');
            });
        }

        async function regenerateImage(promptText) {
            document.getElementById('prompt').value = promptText;
            showToast('Prompt loaded. Click Generate to create new variations.');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function initializeAspectRatio() {
            const firstBtn = document.querySelector('#aspectRatioGroup button');
            if (firstBtn) {
                firstBtn.classList.add('active', 'bg-indigo-600', 'border-indigo-500');
            }
        }

        function initializeSliders() {
            document.getElementById('stepsValue').textContent = '30';
            document.getElementById('guidanceValue').textContent = '7.5';
            document.getElementById('numImagesValue').textContent = '1';
        }

        function initializeTool() {
            initializeTailwind();
            initializeAspectRatio();
            initializeSliders();
            
            const promptInput = document.getElementById('prompt');
            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
                    e.preventDefault();
                    generateImages();
                }
            });

            console.log('%c[AI Image Generator] Tool ready', 'color:#64748b');
        }

        window.onload = initializeTool;
    </script>
