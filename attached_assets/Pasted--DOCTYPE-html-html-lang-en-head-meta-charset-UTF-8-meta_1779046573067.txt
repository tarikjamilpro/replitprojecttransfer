    <script>
        function generateOG() {
            const title = document.getElementById('ogTitle').value || 'Page Title';
            const description = document.getElementById('ogDescription').value || 'Description';
            const url = document.getElementById('ogUrl').value || 'https://yoursite.com';
            const image = document.getElementById('ogImage').value;
            const type = document.getElementById('ogType').value;
            const siteName = document.getElementById('siteName').value;
            const twitterCard = document.getElementById('twitterCard').value;

            let code = `<!-- Open Graph -->\n`;
            code += `<meta property="og:type" content="${type}">\n`;
            code += `<meta property="og:title" content="${title}">\n`;
            code += `<meta property="og:description" content="${description}">\n`;
            code += `<meta property="og:url" content="${url}">\n`;
            if (image) code += `<meta property="og:image" content="${image}">\n`;
            if (siteName) code += `<meta property="og:site_name" content="${siteName}">\n\n`;

            code += `<!-- Twitter -->\n`;
            code += `<meta name="twitter:card" content="${twitterCard}">\n`;
            code += `<meta name="twitter:title" content="${title}">\n`;
            code += `<meta name="twitter:description" content="${description}">\n`;
            if (image) code += `<meta name="twitter:image" content="${image}">\n`;

            document.getElementById('codeOutput').textContent = code;

            // Live Preview
            const preview = document.getElementById('preview');
            preview.innerHTML = `
                <div class="flex gap-4">
                    ${image ? `<img src="${image}" class="w-20 h-20 object-cover rounded">` : ''}
                    <div>
                        <div class="font-semibold">${title}</div>
                        <div class="text-sm text-slate-500">${description}</div>
                        <div class="text-xs text-emerald-600 mt-1">${url}</div>
                    </div>
                </div>
            `;
        }

        function copyCode() {
            const code = document.getElementById('codeOutput').textContent;
            navigator.clipboard.writeText(code).then(() => alert("Copied to clipboard!"));
        }

        ['ogTitle','ogDescription','ogUrl','ogImage','ogType','siteName','twitterCard'].forEach(id => {
            const el = document.getElementById(id);
            if (el) el.addEventListener('input', generateOG);
        });

        window.onload = generateOG;
    </script>
