    <script>
        // ============================================
        // CONFIGURATION
        // ============================================
        let selectedNumImages = 2;
        let selectedRatio = { name: "1:1", width: 1024, height: 1024 };

        const aspectRatios = [
            { name: "1:1", width: 1024, height: 1024 },
            { name: "16:9", width: 1280, height: 720 },
            { name: "9:16", width: 720, height: 1280 },
            { name: "4:3", width: 1024, height: 768 }
        ];

        // ============================================
        // TOAST NOTIFICATION
        // ============================================
        function showToast(message, type = 'success') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            
            const bg = type === 'success' ? 'bg-emerald-600' : 'bg-red-600';
            const icon = type === 'success' ? 'fa-check-circle' : 'fa-exclamation-circle';
            
            toast.className = `toast flex items-center gap-x-3 px-5 py-3 rounded-2xl text-white shadow-xl ${bg}`;
            toast.innerHTML = `
                <i class="fas ${icon}"></i>
                <span class="text-sm font-medium">${message}</span>
            `;
            
            container.appendChild(toast);
            
            setTimeout(() => {
                toast.style.transition = 'all 0.3s ease';
                toast.style.opacity = '0';
                setTimeout(() => toast.remove(), 300);
            }, 2500);
        }

        // ============================================
        // CREATE NUMBER OF IMAGES BUTTONS
        // ============================================
        function createNumImagesButtons() {
            const container = document.getElementById('numImagesContainer');
            container.innerHTML = '';
            
            for (let i = 1; i <= 4; i++) {
                const btn = document.createElement('button');
                btn.textContent = i;
                btn.className = `option-btn flex-1 py-3 rounded-2xl font-medium border border-zinc-700 ${i === selectedNumImages ? 'active bg-violet-600 border-violet-600' : 'bg-zinc-800 hover:bg-zinc-700'}`;
                
                btn.onclick = () => {
                    selectedNumImages = i;
                    createNumImagesButtons(); // refresh active state
                };
                
                container.appendChild(btn);
            }
        }

        // ============================================
        // CREATE ASPECT RATIO BUTTONS
        // ============================================
        function createAspectRatioButtons() {
            const container = document.getElementById('aspectRatioContainer');
            container.innerHTML = '';
            
            aspectRatios.forEach(ratio => {
                const btn = document.createElement('button');
                btn.textContent = ratio.name;
                btn.className = `option-btn px-5 py-3 rounded-2xl font-medium border border-zinc-700 ${ratio.name === selectedRatio.name ? 'active bg-violet-600 border-violet-600' : 'bg-zinc-800 hover:bg-zinc-700'}`;
                
                btn.onclick = () => {
                    selectedRatio = ratio;
                    createAspectRatioButtons();
                };
                
                container.appendChild(btn);
            });
        }

        // ============================================
        // GENERATE IMAGES
        // ============================================
        async function generateImages() {
            const prompt = document.getElementById('prompt').value.trim();
            const negativePrompt = document.getElementById('negativePrompt').value.trim();

            if (!prompt) {
                showToast("Please enter a prompt", "error");
                return;
            }

            const loading = document.getElementById('loading');
            const resultsGrid = document.getElementById('resultsGrid');

            loading.classList.remove('hidden');
            resultsGrid.classList.add('hidden');
            resultsGrid.innerHTML = '';

            try {
                const promises = [];
                
                for (let i = 0; i < selectedNumImages; i++) {
                    promises.push(
                        puter.ai.txt2img(prompt, {
                            model: "wan-ai/wan2.6-image",
                            negative_prompt: negativePrompt || undefined,
                            width: selectedRatio.width,
                            height: selectedRatio.height,
                            quality: "high"
                        })
                    );
                }

                const images = await Promise.all(promises);
                
                images.forEach((img, index) => {
                    const card = createImageCard(img.src, prompt, index);
                    resultsGrid.appendChild(card);
                });

                resultsGrid.classList.remove('hidden');
                showToast(`${images.length} image(s) generated successfully!`);

            } catch (error) {
                console.error(error);
                showToast("Failed to generate images. Please try again.", "error");
            } finally {
                loading.classList.add('hidden');
            }
        }

        // ============================================
        // CREATE IMAGE CARD
        // ============================================
        function createImageCard(imageSrc, prompt, index) {
            const card = document.createElement('div');
            card.className = `image-card bg-zinc-900 border border-zinc-800 rounded-3xl overflow-hidden`;
            
            card.innerHTML = `
                <img src="${imageSrc}" class="w-full aspect-square object-cover" alt="Generated image ${index + 1}">
                
                <div class="p-4 space-y-3">
                    <div class="flex gap-2">
                        <a href="${imageSrc}" download="wan-ai-${Date.now()}-${index + 1}.png"
                           class="flex-1 flex items-center justify-center gap-x-2 bg-emerald-600 hover:bg-emerald-700 py-2.5 rounded-2xl text-sm font-medium transition-colors">
                            <i class="fas fa-download"></i>
                            <span>Download</span>
                        </a>
                        
                        <button onclick="copyPrompt('${prompt.replace(/'/g, "\\'")}')"
                                class="flex-1 flex items-center justify-center gap-x-2 bg-zinc-700 hover:bg-zinc-600 py-2.5 rounded-2xl text-sm font-medium transition-colors">
                            <i class="fas fa-copy"></i>
                            <span>Copy Prompt</span>
                        </button>
                    </div>
                    
                    <button onclick="regenerateSingleImage(this, '${prompt.replace(/'/g, "\\'")}')"
                            class="w-full flex items-center justify-center gap-x-2 bg-zinc-800 hover:bg-zinc-700 py-2.5 rounded-2xl text-sm font-medium transition-colors">
                        <i class="fas fa-redo"></i>
                        <span>Regenerate</span>
                    </button>
                </div>
            `;
            
            return card;
        }

        // ============================================
        // COPY PROMPT
        // ============================================
        function copyPrompt(promptText) {
            navigator.clipboard.writeText(promptText).then(() => {
                showToast("Prompt copied to clipboard!");
            });
        }

        // ============================================
        // REGENERATE SINGLE IMAGE
        // ============================================
        async function regenerateSingleImage(button, originalPrompt) {
            const card = button.closest('.image-card');
            const img = card.querySelector('img');
            
            button.innerHTML = `<i class="fas fa-spinner fa-spin"></i> <span>Regenerating...</span>`;
            button.disabled = true;

            try {
                const result = await puter.ai.txt2img(originalPrompt, {
                    model: "wan-ai/wan2.6-image",
                    width: selectedRatio.width,
                    height: selectedRatio.height,
                    quality: "high"
                });

                img.src = result.src;
                showToast("Image regenerated!");

            } catch (error) {
                showToast("Regeneration failed", "error");
            } finally {
                button.innerHTML = `<i class="fas fa-redo"></i> <span>Regenerate</span>`;
                button.disabled = false;
            }
        }

        // ============================================
        // KEYBOARD SUPPORT
        // ============================================
        function setupKeyboardSupport() {
            const promptInput = document.getElementById('prompt');
            
            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
                    e.preventDefault();
                    generateImages();
                }
            });
        }

        // ============================================
        // INITIALIZATION
        // ============================================
        function initializeTool() {
            createNumImagesButtons();
            createAspectRatioButtons();
            setupKeyboardSupport();
            
            // Set default active states
            console.log('%c[Wan AI Generator] Tool initialized successfully', 'color:#64748b');
        }

        window.onload = initializeTool;
    </script>
