  <script>
    let selectedModel = "flux-schnell";
    let selectedNum = 1;
    let selectedWidth = 1024;
    let selectedHeight = 1024;
    let isGenerating = false;

    function selectModel(element, model) {
      document.querySelectorAll('.model-btn').forEach(btn => btn.classList.remove('bg-blue-600', 'text-white'));
      element.classList.add('bg-blue-600', 'text-white');
      selectedModel = model;

      const note = document.getElementById('modelNote');
      if (model.includes('wan')) {
        note.innerHTML = '<span class="text-amber-400">Wan is high quality but slower (15-40s per image)</span>';
      } else {
        note.innerHTML = 'Fast mode recommended for quick results';
        note.className = 'text-xs text-emerald-400 mt-2';
      }
    }

    function selectNum(element, num) {
      document.querySelectorAll('#numImages button').forEach(b => b.classList.remove('bg-blue-600'));
      element.classList.add('bg-blue-600');
      selectedNum = num;
    }

    function selectAspect(element, w, h) {
      document.querySelectorAll('#aspectRatio button').forEach(b => b.classList.remove('bg-blue-600'));
      element.classList.add('bg-blue-600');
      selectedWidth = w;
      selectedHeight = h;
    }

    let currentGeneration = null;

    async function generateImages() {
      const prompt = document.getElementById('prompt').value.trim();
      const negative = document.getElementById('negativePrompt').value.trim();

      if (!prompt) {
        alert("Please enter a prompt!");
        return;
      }

      isGenerating = true;
      document.getElementById('loadingState').classList.remove('hidden');
      document.getElementById('resultSection').classList.add('hidden');

      const resultsContainer = document.getElementById('results');
      resultsContainer.innerHTML = '';

      const loadingTitle = document.getElementById('loadingTitle');
      const loadingText = document.getElementById('loadingText');
      const progressText = document.getElementById('progressText');

      try {
        for (let i = 0; i < selectedNum; i++) {
          if (!isGenerating) break;

          progressText.innerHTML = `Generating image ${i + 1} of ${selectedNum}...`;

          const controller = new AbortController();
          currentGeneration = controller;

          const timeout = setTimeout(() => {
            controller.abort();
          }, 90000); // 90 second timeout

          try {
            const image = await puter.ai.txt2img(prompt, {
              model: selectedModel,
              negative_prompt: negative || undefined,
              width: selectedWidth,
              height: selectedHeight
            });

            clearTimeout(timeout);

            // Create result card
            const card = document.createElement('div');
            card.className = "bg-zinc-900 border border-zinc-800 rounded-3xl overflow-hidden";
            card.innerHTML = `
              <img src="${image.src}" class="w-full" alt="Generated image">
              <div class="p-4 flex gap-3">
                <a href="${image.src}" download="generated-${Date.now()}.png" 
                   class="flex-1 bg-emerald-600 hover:bg-emerald-700 py-3 rounded-2xl text-center font-medium">
                  <i class="fas fa-download"></i> Download
                </a>
                <button onclick="this.closest('.bg-zinc-900').remove()" 
                   class="flex-1 bg-zinc-700 hover:bg-zinc-600 py-3 rounded-2xl text-center font-medium">
                  <i class="fas fa-trash"></i> Remove
                </button>
              </div>
            `;
            resultsContainer.appendChild(card);

          } catch (err) {
            clearTimeout(timeout);
            if (err.name === 'AbortError') {
              alert(`Image ${i+1} timed out. Try a faster model or reduce number of images.`);
            } else {
              alert(`Error generating image ${i+1}: ${err.message}`);
            }
          }
        }

        document.getElementById('loadingState').classList.add('hidden');
        document.getElementById('resultSection').classList.remove('hidden');

      } catch (error) {
        alert("Something went wrong: " + error.message);
        document.getElementById('loadingState').classList.add('hidden');
      }

      isGenerating = false;
      currentGeneration = null;
    }

    function cancelGeneration() {
      isGenerating = false;
      if (currentGeneration) {
        currentGeneration.abort();
      }
      document.getElementById('loadingState').classList.add('hidden');
      alert("Generation cancelled.");
    }

    function clearResults() {
      document.getElementById('results').innerHTML = '';
      document.getElementById('resultSection').classList.add('hidden');
    }

    // Keyboard shortcut
    document.addEventListener('keydown', function(e) {
      if (e.ctrlKey && e.key === "Enter") {
        generateImages();
      }
    });

    // Set default selections on load
    window.onload = function() {
      // Default to Flux (fast)
      document.querySelector('.model-btn').classList.add('bg-blue-600', 'text-white');
      
      // Default number of images
      document.querySelector('#numImages button').classList.add('bg-blue-600');
      
      // Default aspect ratio
      document.querySelector('#aspectRatio button').classList.add('bg-blue-600');
    }
  </script>
