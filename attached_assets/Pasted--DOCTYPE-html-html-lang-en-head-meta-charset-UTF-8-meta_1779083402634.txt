    <script>
        async function generateContent() {
            const topic = document.getElementById('topicInput').value.trim();
            if (!topic) {
                alert("Please enter a topic or title");
                return;
            }

            const outputSection = document.getElementById('outputSection');
            outputSection.classList.add('hidden');

            // Show loading state
            const generateBtn = event.currentTarget;
            
            try {
                const res = await fetch('/generate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ topic: topic })
                });

                const data = await res.json();

                if (data.error) {
                    alert(data.error);
                    return;
                }

                // Render Titles
                const titlesContainer = document.getElementById('titlesContainer');
                titlesContainer.innerHTML = '';

                data.titles.forEach(title => {
                    const div = document.createElement('div');
                    div.className = `flex justify-between items-center p-3 bg-gray-50 rounded-2xl hover:bg-gray-100`;
                    div.innerHTML = `
                        <span class="flex-1 pr-4">${title}</span>
                        <button onclick="copySingleTitle(this)" 
                                class="text-purple-600 hover:text-purple-700 text-sm px-3 py-1 transition-colors">
                            Copy
                        </button>
                    `;
                    titlesContainer.appendChild(div);
                });

                // Render other sections
                document.getElementById('tagsOutput').innerText = data.seo_tags;
                document.getElementById('hashtagsOutput').innerText = data.hashtags;
                document.getElementById('captionOutput').innerText = data.caption;

                outputSection.classList.remove('hidden');

            } catch (error) {
                alert("Something went wrong. Please try again.");
                console.error(error);
            }
        }

        function copySingleTitle(element) {
            const text = element.parentElement.innerText.replace('Copy', '').trim();
            navigator.clipboard.writeText(text);
            const originalText = element.innerText;
            element.innerText = "Copied!";
            setTimeout(() => {
                element.innerText = originalText;
            }, 1500);
        }

        function copyElement(id) {
            const text = document.getElementById(id).innerText;
            navigator.clipboard.writeText(text);
            alert("Copied to clipboard!");
        }

        function copyAllTitles() {
            const titles = Array.from(document.querySelectorAll('#titlesContainer span'))
                .map(el => el.innerText)
                .join('\n');
            
            navigator.clipboard.writeText(titles);
            alert("All titles copied to clipboard!");
        }

        // Allow pressing Enter to generate
        document.getElementById('topicInput').addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                generateContent();
            }
        });
    </script>
