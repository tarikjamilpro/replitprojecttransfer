                        </div>
                    </div>

                    <!-- JSON-LD Type -->
                    <div>
                        <label class="text-sm text-slate-600 block mb-1.5">JSON-LD Schema Type</label>
                        <select id="schemaType" onchange="toggleSchemaFields(); generateAll()" class="w-full border border-slate-300 rounded-2xl px-4 py-2.5 bg-white">

                            <option value="Article">Article</option>

    <script>
        function toggleSchemaFields() {
            const type = document.getElementById('schemaType').value;
            const container = document.getElementById('schemaFields');
            container.innerHTML = '';
            container.classList.remove('hidden');

            if (type === 'Article') {
                container.innerHTML = `
                    <div><label class="text-sm">Date Published</label><input type="date" id="datePublished" class="w-full border rounded-2xl px-4 py-2"></div>
                    <div><label class="text-sm">Image URL</label><input type="url" id="articleImage" class="w-full border rounded-2xl px-4 py-2"></div>
                `;
            } 
            else if (type === 'Product') {
                container.innerHTML = `
                    <div class="grid grid-cols-2 gap-4">
                        <div><label class="text-sm">Price</label><input type="number" id="price" class="w-full border rounded-2xl px-4 py-2"></div>
                        <div><label class="text-sm">Currency</label><input type="text" id="currency" value="USD" class="w-full border rounded-2xl px-4 py-2"></div>
                    </div>
                    <div><label class="text-sm">Brand</label><input type="text" id="brand" class="w-full border rounded-2xl px-4 py-2"></div>
                `;
            } 
            else if (type === 'WebSite') {
                container.innerHTML = `<p class="text-sm text-slate-500">Basic Website schema will be generated.</p>`;
            } 
            else {
                container.classList.add('hidden');
            }
        }

        function generateAll() {
            const title = document.getElementById('title').value || 'Page Title';
            const description = document.getElementById('description').value || 'Page description';
            const keywords = document.getElementById('keywords').value;
            const author = document.getElementById('author').value;
            const canonical = document.getElementById('canonical').value;
            const schemaType = document.getElementById('schemaType').value;

            let output = `<!-- Primary Meta Tags -->\n`;
            output += `<title>${title}</title>\n`;
            output += `<meta name="description" content="${description}">\n`;
            if (keywords) output += `<meta name="keywords" content="${keywords}">\n`;
            if (canonical) output += `<link rel="canonical" href="${canonical}">\n\n`;

            output += `<!-- Open Graph -->\n`;
            output += `<meta property="og:title" content="${title}">\n`;
            output += `<meta property="og:description" content="${description}">\n`;
            if (canonical) output += `<meta property="og:url" content="${canonical}">\n\n`;

            output += `<!-- Twitter -->\n`;
            output += `<meta name="twitter:card" content="summary_large_image">\n`;
            output += `<meta name="twitter:title" content="${title}">\n`;
            output += `<meta name="twitter:description" content="${description}">\n`;

            // JSON-LD
            if (schemaType) {
                let jsonLD = {};
                const base = {
                    "@context": "https://schema.org",
                    "@type": schemaType,
                    "name": title,
                    "description": description,
                    "url": canonical || window.location.href
                };

                if (schemaType === "Article") {
                    jsonLD = {
                        ...base,
                        "headline": title,
                        "author": { "@type": "Person", "name": author || "Author Name" },
                        "datePublished": document.getElementById('datePublished')?.value || new Date().toISOString(),
                        "image": document.getElementById('articleImage')?.value || ""
                    };
                } 
                else if (schemaType === "Product") {
                    jsonLD = {
                        ...base,
                        "brand": { "@type": "Brand", "name": document.getElementById('brand')?.value || "Brand" },
                        "offers": {
                            "@type": "Offer",
                            "priceCurrency": document.getElementById('currency')?.value || "USD",
                            "price": document.getElementById('price')?.value || "0"
                        }
                    };
                }

                output += `\n<!-- JSON-LD Structured Data -->\n`;
                output += `<script type="application/ld+json">\n${JSON.stringify(jsonLD, null, 2)}\n<\/script>`;
            }

            document.getElementById('output').textContent = output;
        }

        function copyAll() {
            const text = document.getElementById('output').textContent;
            navigator.clipboard.writeText(text).then(() => alert("Copied to clipboard!"));
        }

        // Auto update
        window.onload = () => {
            ['title','description','keywords','author','canonical','schemaType'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.addEventListener('input', generateAll);
            });
            generateAll();
        };
    </script>
