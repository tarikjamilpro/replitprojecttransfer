import { Request, Response } from 'express';
import pool from '../config/db';

// Get all prompts (Public)
export const getAllPrompts = async (req: Request, res: Response) => {
  try {
    const result = await pool.query(
      'SELECT * FROM ai_prompts ORDER BY created_at DESC'
    );
    res.json(result.rows);
  } catch (error) {
    console.error(error);
    res.status(500).json({ message: 'Failed to fetch prompts' });
  }
};

// Get single prompt by ID
export const getPromptById = async (req: Request, res: Response) => {
  const { id } = req.params;
  try {
    const result = await pool.query(
      'SELECT * FROM ai_prompts WHERE id = $1',
      [id]
    );
    if (result.rows.length === 0) {
      return res.status(404).json({ message: 'Prompt not found' });
    }
    res.json(result.rows[0]);
  } catch (error) {
    res.status(500).json({ message: 'Failed to fetch prompt' });
  }
};

// Create new prompt (Admin only)
export const createPrompt = async (req: Request, res: Response) => {
  const { title, description, image_url, meigen_target_link } = req.body;

  if (!title || !image_url || !meigen_target_link) {
    return res.status(400).json({ message: 'Title, Image URL and Target Link are required' });
  }

  try {
    const result = await pool.query(
      `INSERT INTO ai_prompts (title, description, image_url, meigen_target_link) 
       VALUES ($1, $2, $3, $4) RETURNING *`,
      [title, description, image_url, meigen_target_link]
    );
    res.status(201).json(result.rows[0]);
  } catch (error) {
    console.error(error);
    res.status(500).json({ message: 'Failed to create prompt' });
  }
};

// Update prompt (Admin only)
export const updatePrompt = async (req: Request, res: Response) => {
  const { id } = req.params;
  const { title, description, image_url, meigen_target_link } = req.body;

  try {
    const result = await pool.query(
      `UPDATE ai_prompts 
       SET title = $1, description = $2, image_url = $3, meigen_target_link = $4, updated_at = NOW()
       WHERE id = $5 RETURNING *`,
      [title, description, image_url, meigen_target_link, id]
    );

    if (result.rows.length === 0) {
      return res.status(404).json({ message: 'Prompt not found' });
    }

    res.json(result.rows[0]);
  } catch (error) {
    res.status(500).json({ message: 'Failed to update prompt' });
  }
};

// Delete prompt (Admin only)
export const deletePrompt = async (req: Request, res: Response) => {
  const { id } = req.params;

  try {
    const result = await pool.query(
      'DELETE FROM ai_prompts WHERE id = $1 RETURNING *',
      [id]
    );

    if (result.rows.length === 0) {
      return res.status(404).json({ message: 'Prompt not found' });
    }

    res.json({ message: 'Prompt deleted successfully' });
  } catch (error) {
    res.status(500).json({ message: 'Failed to delete prompt' });
  }
};
