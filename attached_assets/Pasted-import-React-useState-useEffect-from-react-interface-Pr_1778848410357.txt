import React, { useState, useEffect } from 'react';

interface Prompt {
  id: number;
  title: string;
  description: string;
  image_url: string;
  meigen_target_link: string;
  created_at: string;
}

const PromptManager: React.FC = () => {
  const [prompts, setPrompts] = useState<Prompt[]>([]);
  const [filteredPrompts, setFilteredPrompts] = useState<Prompt[]>([]);
  const [searchTerm, setSearchTerm] = useState('');
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingPrompt, setEditingPrompt] = useState<Prompt | null>(null);
  const [loading, setLoading] = useState(true);

  // Form state
  const [formData, setFormData] = useState({
    title: '',
    description: '',
    image_url: '',
    meigen_target_link: '',
  });

  // Fetch all prompts
  const fetchPrompts = async () => {
    try {
      const res = await fetch('/api/prompts');
      const data = await res.json();
      setPrompts(data);
      setFilteredPrompts(data);
    } catch (error) {
      console.error('Failed to fetch prompts');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchPrompts();
  }, []);

  // Search functionality
  useEffect(() => {
    const filtered = prompts.filter(
      (prompt) =>
        prompt.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
        prompt.description.toLowerCase().includes(searchTerm.toLowerCase())
    );
    setFilteredPrompts(filtered);
  }, [searchTerm, prompts]);

  // Open modal for add or edit
  const openModal = (prompt?: Prompt) => {
    if (prompt) {
      setEditingPrompt(prompt);
      setFormData({
        title: prompt.title,
        description: prompt.description,
        image_url: prompt.image_url,
        meigen_target_link: prompt.meigen_target_link,
      });
    } else {
      setEditingPrompt(null);
      setFormData({ title: '', description: '', image_url: '', meigen_target_link: '' });
    }
    setIsModalOpen(true);
  };

  const closeModal = () => {
    setIsModalOpen(false);
    setEditingPrompt(null);
  };

  // Handle form input
  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
  };

  // Save (Create or Update)
  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    const url = editingPrompt 
      ? `/api/prompts/${editingPrompt.id}` 
      : '/api/prompts';
    
    const method = editingPrompt ? 'PUT' : 'POST';

    try {
      const res = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData),
      });

      if (res.ok) {
        fetchPrompts();
        closeModal();
      }
    } catch (error) {
      console.error('Failed to save prompt');
    }
  };

  // Delete prompt
  const handleDelete = async (id: number) => {
    if (!window.confirm('Are you sure you want to delete this prompt?')) return;

    try {
      await fetch(`/api/prompts/${id}`, { method: 'DELETE' });
      fetchPrompts();
    } catch (error) {
      console.error('Failed to delete prompt');
    }
  };

  return (
    <div className="p-6">
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-2xl font-semibold">Prompt Manager</h2>
        <button
          onClick={() => openModal()}
          className="flex items-center gap-x-2 px-5 py-2.5 bg-violet-600 hover:bg-violet-700 rounded-2xl text-sm font-medium"
        >
          <i className="fas fa-plus"></i>
          <span>Add New Prompt</span>
        </button>
      </div>

      {/* Search */}
      <div className="mb-6">
        <input
          type="text"
          placeholder="Search prompts..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
          className="w-full md:w-96 bg-zinc-900 border border-zinc-700 focus:border-violet-500 rounded-2xl px-5 py-3 text-sm"
        />
      </div>

      {/* Table */}
      <div className="bg-zinc-900 border border-zinc-800 rounded-3xl overflow-hidden">
        <table className="w-full">
          <thead className="bg-zinc-950">
            <tr className="text-left text-xs text-zinc-400">
              <th className="px-6 py-4">Title</th>
              <th className="px-6 py-4 hidden lg:table-cell">Description</th>
              <th className="px-6 py-4">MeiGen Link</th>
              <th className="px-6 py-4 text-right">Actions</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-zinc-800">
            {filteredPrompts.length > 0 ? (
              filteredPrompts.map((prompt) => (
                <tr key={prompt.id} className="hover:bg-zinc-800/50">
                  <td className="px-6 py-4 font-medium">{prompt.title}</td>
                  <td className="px-6 py-4 text-zinc-400 hidden lg:table-cell line-clamp-2">
                    {prompt.description}
                  </td>
                  <td className="px-6 py-4">
                    <a 
                      href={prompt.meigen_target_link} 
                      target="_blank" 
                      className="text-violet-400 hover:underline text-sm"
                    >
                      View
                    </a>
                  </td>
                  <td className="px-6 py-4 text-right space-x-2">
                    <button 
                      onClick={() => openModal(prompt)}
                      className="px-4 py-1.5 text-xs bg-zinc-700 hover:bg-zinc-600 rounded-lg"
                    >
                      Edit
                    </button>
                    <button 
                      onClick={() => handleDelete(prompt.id)}
                      className="px-4 py-1.5 text-xs bg-red-600/80 hover:bg-red-600 rounded-lg"
                    >
                      Delete
                    </button>
                  </td>
                </tr>
              ))
            ) : (
              <tr>
                <td colSpan={4} className="px-6 py-8 text-center text-zinc-400">
                  No prompts found.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>

      {/* Add/Edit Modal */}
      {isModalOpen && (
        <div className="fixed inset-0 bg-black/70 flex items-center justify-center z-50 p-4">
          <
