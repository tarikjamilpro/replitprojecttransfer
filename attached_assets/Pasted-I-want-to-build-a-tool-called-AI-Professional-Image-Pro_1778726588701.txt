    <script>
        function initializeTailwind() {
            console.log('%c[AI Prompt Generator] Initialized', 'color:#64748b');
        }

        async function generateProfessionalPrompt() {
            const idea = document.getElementById('basicIdea').value.trim();
            const model = document.getElementById('modelSelect').value;
            const generateBtn = document.getElementById('generateBtn');
            const btnText = document.getElementById('btnText');
            const btnIcon = document.getElementById('btnIcon');
            const outputPlaceholder = document.getElementById('outputPlaceholder');
            const outputContent = document.getElementById('outputContent');
            const copyBtn = document.getElementById('copyBtn');

            if (!idea) {
                showError("Please enter a basic idea for your image.");
                return;
            }

            generateBtn.disabled = true;
            btnText.textContent = 'Generating...';
            btnIcon.className = 'fas fa-spinner loading-spinner';
            
            outputPlaceholder.classList.add('hidden');
            outputContent.classList.add('hidden');
            copyBtn.classList.add('hidden');

            try {
                const response = await fetch('/api/enhance-prompt', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ idea: idea, model: model })
                });

                if (!response.ok) throw new Error(`Server error: ${response.status}`);

                const data = await response.json();

                if (!data.prompt) throw new Error('No prompt received from server');

                outputContent.innerHTML = data.prompt;
                outputContent.classList.remove('hidden');
                copyBtn.classList.remove('hidden');

            } catch (error) {
                console.error(error);
                outputPlaceholder.classList.remove('hidden');
                outputPlaceholder.innerHTML = `
                    <div class="text-red-400">
                        <i class="fas fa-exclamation-triangle text-2xl mb-3"></i>
                        <p class="font-medium">Failed to generate prompt</p>
                        <p class="text-xs mt-1 text-red-400/70">${error.message}</p>
                    </div>
                `;
            } finally {
                generateBtn.disabled = false;
                btnText.textContent = 'Generate Professional Prompt';
                btnIcon.className = 'fas fa-wand-magic-sparkles';
            }
        }

        async function copyPromptToClipboard() {
            const outputContent = document.getElementById('outputContent');
            const copyBtn = document.getElementById('copyBtn');
            const copyStatus = document.getElementById('copyStatus');

            if (!outputContent.textContent.trim()) return;

            try {
                await navigator.clipboard.writeText(outputContent.textContent.trim());
                
                const originalText = copyBtn.innerHTML;
                copyBtn.innerHTML = `<i class="fas fa-check text-emerald-400"></i> <span class="text-emerald-400">Copied Successfully!</span>`;
                copyStatus.classList.remove('hidden');
                
                setTimeout(() => {
                    copyBtn.innerHTML = originalText;
                    copyStatus.classList.add('hidden');
                }, 2200);
            } catch (err) {
                const textArea = document.createElement('textarea');
                textArea.value = outputContent.textContent.trim();
                document.body.appendChild(textArea);
                textArea.select();
                document.execCommand('copy');
                document.body.removeChild(textArea);
                
                copyBtn.innerHTML = `<i class="fas fa-check"></i> <span>Copied!</span>`;
                setTimeout(() => {
                    copyBtn.innerHTML = `<i class="fas fa-copy"></i> <span>Copy Prompt to Clipboard</span>`;
                }, 2000);
            }
        }

        function showError(message) {
            const outputPlaceholder = document.getElementById('outputPlaceholder');
            const outputContent = document.getElementById('outputContent');
            const copyBtn = document.getElementById('copyBtn');

            outputContent.classList.add('hidden');
            copyBtn.classList.add('hidden');
            outputPlaceholder.classList.remove('hidden');
            outputPlaceholder.innerHTML = `
                <div class="text-amber-400">
                    <i class="fas fa-exclamation-circle text-xl mb-2"></i>
                    <p class="text-sm">${message}</p>
                </div>
            `;
        }

        function setupKeyboardSupport() {
            const textarea = document.getElementById('basicIdea');
            textarea.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
                    e.preventDefault();
                    generateProfessionalPrompt();
                }
            });
        }

        function initializeTool() {
            initializeTailwind();
            setupKeyboardSupport();
        }

        window.onload = initializeTool;
    </script>
